
let points = [];
let current = { x: 0, y: 0 };

let commands = [];
let cmdIndex = 0;

let moving = false;
let startPos, targetPos;

let progress = 0;
let speed = 0.02;

// animated relative
let relX = 0;
let relY = 0;

// 📷 camera
let cam = { x: 0, y: 0 };

// 🔍 zoom
let zoom = 1;

function setup() {
  createCanvas(600, 600);
  frameRate(60);

  commands = [
    { type: "ABS", x: 200, y: 200 },

    { type: "REL", dx: 30, dy: 0 },
    { type: "REL", dx: 0, dy: 30 },
    { type: "REL", dx: 150, dy: 0 },
    { type: "REL", dx: 7.5, dy: 7.5 },
    { type: "REL", dx: 0, dy: 75 },
    { type: "REL", dx: -7.5, dy: 7.5 },
    { type: "REL", dx: -150, dy: 0 },
    { type: "REL", dx: 0, dy: 30 },
    { type: "REL", dx: -30, dy: 0 },

    { type: "ABS", x: 200, y: 200 }
  ];

  points.push({ ...current });
}

function draw() {
  background(240);

  // 📷 Camera follow
  cam.x = lerp(cam.x, current.x, 0.1);
  cam.y = lerp(cam.y, current.y, 0.1);

  translate(width / 2, height / 2);

  // 🔍 Apply zoom + camera
  scale(zoom, -zoom);
  translate(-cam.x, -cam.y);

  drawGrid(50);

  // 🔷 Path
  stroke(0);
  noFill();
  beginShape();
  for (let p of points) {
    vertex(p.x, p.y);
  }
  endShape();

  // 🔴 Point
  fill("red");
  noStroke();
  circle(current.x, current.y, 10 / zoom);

  // 🧭 Local axes
  push();
  translate(current.x, current.y);

  strokeWeight(2 / zoom);

  stroke("red");
  line(0, 0, 50, 0);

  stroke("green");
  line(0, 0, 0, 50);

  pop();

  // 📝 Text at point (fixed size)
  scale(1, -1);
  fill(0);
  textSize(12 / zoom);
  text(
    `(${current.x.toFixed(1)}, ${current.y.toFixed(1)})\n@(${relX.toFixed(1)}, ${relY.toFixed(1)})`,
    current.x + 10,
    -current.y - 10
  );
  scale(1, -1);

  updateMotion();
}

function updateMotion() {
  if (!moving && cmdIndex < commands.length) {
    startPos = { ...current };

    let cmd = commands[cmdIndex];

    if (cmd.type === "ABS") {
      targetPos = { x: cmd.x, y: cmd.y };
    } else {
      targetPos = {
        x: current.x + cmd.dx,
        y: current.y + cmd.dy
      };
    }

    progress = 0;
    relX = 0;
    relY = 0;
    moving = true;
  }

  if (moving) {
    progress += speed;

    let t = constrain(progress, 0, 1);

    current.x = lerp(startPos.x, targetPos.x, t);
    current.y = lerp(startPos.y, targetPos.y, t);

    relX = current.x - startPos.x;
    relY = current.y - startPos.y;

    if (t >= 1) {
      current = { ...targetPos };
      points.push({ ...current });

      moving = false;
      cmdIndex++;
    }
  }
}

// 🔲 Grid (scales correctly)
function drawGrid(size) {
  stroke(220);
  strokeWeight(1 / zoom);

  for (let x = -3000; x <= 3000; x += size) {
    line(x, -3000, x, 3000);
  }

  for (let y = -3000; y <= 3000; y += size) {
    line(-3000, y, 3000, y);
  }
}

// 🔍 Mouse wheel zoom
function mouseWheel(event) {
  let zoomFactor = 1.05;

  if (event.delta > 0) {
    zoom /= zoomFactor; // zoom out
  } else {
    zoom *= zoomFactor; // zoom in
  }

  zoom = constrain(zoom, 0.2, 5);

  return false;
}
